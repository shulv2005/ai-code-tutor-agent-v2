# =============================================================================
# AI 代码导师系统 —— 运行期依赖清单
#
# 用途：start.bat 会自动执行 `pip install -r requirements.txt` 安装以下依赖，
#       学生双击一个 .bat 就能把整套系统跑起来，不需要懂 pip。
#
# 覆盖的功能模块（详见 README.md）：
#   · AI 自动检测（语法+深度） app/services/code_checker.py
#   · AI 代码改错            app/services/code_fixer.py
#   · AI 注释生成            app/services/comment_generator.py
#   · 错题本 + 举一反三       app/services/tutor_service.py、app/agents/tutor_agent.py
#   · 多语言代码解析          app/services/repo/（tree-sitter，只保留解析）
#
# 注意：本文件必须与 pyproject.toml 的 [project.dependencies] 保持一致，
#       否则 `pip install -r requirements.txt` 与 `pip install -e .` 装出来的环境会不同。
#       可以用 `python tools/verify_integration.py` 自动比对两者。
#
# 【已移除的重依赖】原先「开源贡献端」（克隆仓库 / 混合检索 / 生成测试 / Docker 沙箱 /
#   自动修复 / PR 草稿）用到的这些包已随该功能整体删除，安装体积从几百 MB 降到几十 MB：
#       GitPython / rank_bm25 / faiss-cpu / numpy / fastembed / jieba / docker
# =============================================================================

# ---------- Web 后端框架（FastAPI 及其运行依赖）----------
fastapi>=0.115.0
uvicorn[standard]>=0.30.0
pydantic>=2.7.0
pydantic-settings>=2.3.0
python-dotenv>=1.0.1
# 前端上传代码文件用（FastAPI 处理 multipart/form-data 必需，缺了会直接报错）
python-multipart>=0.0.9

# ---------- 数据库（本地 SQLite，异步驱动）----------
# 错题本就存在这个 SQLite 文件里（data/app.db），不需要安装任何数据库服务。
# 想换成 MySQL / PostgreSQL，只要改 .env 里的 DATABASE__URL_OVERRIDE 并装对应驱动，
# 业务代码一行都不用动（建表 SQL 两种方言都写在 docs/student_error_logs.sql）。
sqlalchemy>=2.0.30
aiosqlite>=0.20.0
# greenlet 必须**显式**写出来，不能指望 sqlalchemy 帮你带 —— 这是"新电脑上跑不起来"
# 最常见的那个坑。SQLAlchemy 的 async 引擎靠它，但它对它的依赖标记是
#   greenlet>=1; platform_machine == "aarch64" or ... "x86_64" or ...
# 在 ARM64 这类不在名单里的机器上 pip 会**跳过** greenlet：
# sqlalchemy 装得上、import 也不报错，直到真正建异步引擎才炸出
#   ModuleNotFoundError: No module named 'greenlet'
# 使用者看到的就是双击 start.bat 后一屏英文堆栈。
greenlet>=3.0.0

# ---------- HTTP 客户端（调用大模型 API）----------
httpx>=0.27.0
# 通用 LLM 客户端（app/core/llm_client.py）：官方 SDK，负责鉴权头、超时与
# SSE 流式解析；重试与错误映射由项目自管（见该文件模块注释）
openai>=1.40.0

# ---------- 多语言代码解析（本地语法检查用）----------
tree-sitter>=0.23
# 注意：语法包必须用「自带编译产物」的独立 wheel（tree-sitter-python 等），
# 不要换成 tree-sitter-language-pack —— 它会在运行时从 GitHub 下载语法库与
# parsers.json，内网 / 离线环境会直接下载超时失败。
# AI 代码导师面向 C / Java / Python 学习者，因此额外装上 c 与 java 语法包。
tree-sitter-python>=0.23
tree-sitter-javascript>=0.23
tree-sitter-go>=0.23
tree-sitter-c>=0.23
tree-sitter-java>=0.23
